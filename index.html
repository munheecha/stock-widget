async function getAccessToken(env) { const kvKey = "kis_access_token"; // 1. KV에 캐싱된 토큰이 있는지 확인 const cachedToken = await env.ALERTS_KV.get(kvKey); if (cachedToken) { return cachedToken; } // 2. 없으면 새로 발급 요청 const tokenRes = await fetch( "https://openapi.koreainvestment.com:9443/oauth2/tokenP", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ grant_type: "client_credentials", appkey: env.KIS_APP_KEY, appsecret: env.KIS_APP_SECRET }) } ); const tokenData = await tokenRes.json(); if (!tokenData.access_token) { throw new Error("토큰 발급 실패: " + JSON.stringify(tokenData)); } const accessToken = tokenData.access_token; // 3. 발급받은 토큰을 KV에 12시간(43200초) 동안 캐싱 (토큰 재요청 방지) await env.ALERTS_KV.put(kvKey, accessToken, { expirationTtl: 43200 }); return accessToken; } async function runStockCheck(env) { // 캐싱된 토큰 가져오기 사용 const accessToken = await getAccessToken(env); const stockList = [ { name: "삼성전자", symbol: "005930.KS", code: "005930" }, { name: "SK하이닉스", symbol: "000660.KS", code: "000660" }, { name: "KODEX 미국나스닥100", symbol: "379810.KS", code: "379810" }, { name: "일진전기", symbol: "103590.KS", code: "103590" } ]; const result = []; for (const stock of stockList) { const priceRes = await fetch( "https://openapi.koreainvestment.com:9443/uapi/domestic-stock/v1/quotations/inquire-price?FID_COND_MRKT_DIV_CODE=J&FID_INPUT_ISCD=" + stock.code, { method: "GET", headers: { authorization: "Bearer " + accessToken, appkey: env.KIS_APP_KEY, appsecret: env.KIS_APP_SECRET, tr_id: "FHKST01010100", custtype: "P" } } ); const priceData = await priceRes.json(); // KV에서 목표가 조회 const targetValue = await env.ALERTS_KV.get(`target_${stock.symbol}`); const target = Number(targetValue) || 0; if (priceData.output && priceData.output.stck_prpr) { const currentPrice = Number(priceData.output.stck_prpr); // 목표가 도달 시 텔레그램 알림 전송 if (target > 0 && currentPrice >= target) { // 이미 알림을 보냈는지 체크하는 로직이 없으면 매번 울릴 수 있으니 참고용 유지 await fetch( `https://api.telegram.org/bot${env.telegram_bot_token}/sendMessage`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chat_id: env.telegram_chat_id, text: `🎉 목표가 도달\n\n종목: ${stock.name}\n현재가: ${currentPrice.toLocaleString()}원\n목표가: ${target.toLocaleString()}원` }) } ); } result.push({ name: stock.name, symbol: stock.symbol, price: currentPrice, target: target }); } else { result.push({ name: stock.name, symbol: stock.symbol, error: true, target: target }); } } return result; } export default { async fetch(request, env) { if (request.method === "OPTIONS") { return new Response(null, { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" } }); } if (request.method === "POST") { try { const data = await request.json(); await env.ALERTS_KV.put( `target_${data.symbol}`, String(data.target) ); return new Response(JSON.stringify({ success: true }), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } catch (error) { return new Response(JSON.stringify({ success: false, error: error.message }), { status: 400, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } }); } } try { const result = await runStockCheck(env); return new Response( JSON.stringify(result), { headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*", "Cache-Control": "no-store" } } ); } catch (error) { return new Response( JSON.stringify({ error: error.message }), { status: 500, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*" } } ); } }, async scheduled(event, env, ctx) { try { await runStockCheck(env); } catch (error) { console.log(error); } } };
